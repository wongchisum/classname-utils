import { Bench } from "tinybench";

const bench = new Bench({ name: "simple benchmark", time: 1000 });
 
bench.add("Classy fast test", () => {
  classy("test", "test2");
});
 
bench.add("Classy slow test", () => {
  classy([
    "a",
    "b",
    ["c", "d"],
    null,
    { o: false },
    { e: true },
    ["f", ["g", "h"], { i: false }, { j: true }, ["k", "l"]],
    { u: null, s: undefined },
  ]);
});
